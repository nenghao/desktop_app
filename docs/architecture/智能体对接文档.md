## API 概览

### 接口列表

| 接口 | 方法 | 路径 | 说明 |
|------|------|------|------|
| 智能体列表 | GET | `/v2/agent/` | 获取所有智能体的基础信息 |
| 智能体详情 | GET | `/v2/agent/{agent_id}` | 获取指定智能体的完整信息 |

### 统一响应格式

所有接口都返回统一的响应格式：

```typescript
interface StandardResponse<T> {
  success: boolean;      // 请求是否成功
  message: string;       // 响应消息
  data: T | null;        // 响应数据
}
```

async function getAgentList() {

### 步骤 3: 动态加载 UI

```javascript
async function loadAgentUI(agentId, container) {
  // 1. 获取智能体详情
  const agent = await getAgentDetail(agentId);
  
  // 2. 检查是否有 UI
  if (!agent.has_ui || !agent.ui_config) {
    throw new Error('该智能体没有自定义 UI');
  }
  
  const { bundle_url, css_url, config_url } = agent.ui_config;
  
  // 3. 加载 CSS
  await loadCSS(css_url);
  
  // 4. 加载 JS
  await loadScript(bundle_url);
  
  // 5. 获取配置
  const configRes = await fetch(config_url);
  const config = await configRes.json();
  
  // 6. 渲染组件
  const ComponentClass = window[config.entry_component];
  if (!ComponentClass) {
    throw new Error(`组件 ${config.entry_component} 未找到`);
  }
  
  // 7. 创建实例
  const instance = ComponentClass({
    theme: 'light',
    precision: 2,
    showHistory: true,
    onCalculate: (expr, result) => {
      console.log('计算结果:', expr, '=', result);
    },
    onError: (error) => {
      console.error('计算错误:', error);
    }
  });
  
  // 8. 渲染到容器
  instance.render(container);
  
  return instance;
}

// 辅助函数：加载 CSS
function loadCSS(url) {
  return new Promise((resolve, reject) => {
    // 检查是否已加载
    if (document.querySelector(`link[href="${url}"]`)) {
      resolve();
      return;
    }
    
    const link = document.createElement('link');
    link.rel = 'stylesheet';
    link.href = url;
    link.onload = resolve;
    link.onerror = reject;
    document.head.appendChild(link);
  });
}

// 辅助函数：加载 Script
function loadScript(url) {
  return new Promise((resolve, reject) => {
    // 检查是否已加载
    if (document.querySelector(`script[src="${url}"]`)) {
      resolve();
      return;
    }
    
    const script = document.createElement('script');
    script.src = url;
    script.onload = resolve;
    script.onerror = reject;
    document.head.appendChild(script);
  });
}
```

### Electron 主进程集成

```typescript
// main.ts
import { app, BrowserWindow, ipcMain } from 'electron';
import axios from 'axios';

const API_BASE = 'http://192.168.0.103:10089';

// 获取智能体列表
ipcMain.handle('agent:list', async () => {
  try {
    const response = await axios.get(`${API_BASE}/v2/agent/`);
    return response.data;
  } catch (error) {
    console.error('获取智能体列表失败:', error);
    throw error;
  }
});

// 获取智能体详情
ipcMain.handle('agent:detail', async (event, agentId: string) => {
  try {
    const response = await axios.get(`${API_BASE}/v2/agent/${agentId}`);
    return response.data;
  } catch (error) {
    console.error('获取智能体详情失败:', error);
    throw error;
  }
});

// 获取 UI 配置
ipcMain.handle('agent:ui-config', async (event, agentId: string) => {
  try {
    const response = await axios.get(`${API_BASE}/v2/agent/${agentId}/ui`);
    return response.data;
  } catch (error) {
    console.error('获取 UI 配置失败:', error);
    throw error;
  }
});

// 下载智能体文件
ipcMain.handle('agent:download', async (event, url: string, savePath: string) => {
  try {
    const response = await axios.get(url, { responseType: 'arraybuffer' });
    const fs = require('fs');
    fs.writeFileSync(savePath, Buffer.from(response.data));
    return { success: true, path: savePath };
  } catch (error) {
    console.error('下载失败:', error);
    throw error;
  }
});
```

**渲染进程使用**:

```typescript
// renderer.ts
import { ipcRenderer } from 'electron';

// 获取智能体列表
async function getAgentList() {
  const response = await ipcRenderer.invoke('agent:list');
  return response.data.items;
}

// 获取智能体详情
async function getAgentDetail(agentId: string) {
  const response = await ipcRenderer.invoke('agent:detail', agentId);
  return response.data;
}

// 下载智能体
async function downloadAgent(agent: any) {
  if (agent.type === 'python' && agent.lib_urls) {
    for (const url of agent.lib_urls) {
      const fileName = url.split('/').pop();
      const savePath = `./agents/${agent.id}/${fileName}`;
      await ipcRenderer.invoke('agent:download', url, savePath);
    }
  }
}

// 使用示例
const agents = await getAgentList();
const agent = await getAgentDetail('math_calculator');
await downloadAgent(agent);
```

## 错误处理

### 常见错误码

| 状态码 | 说明 | 处理方式 |
|--------|------|---------|
| 200 | 成功 | 正常处理 |
| 404 | 智能体不存在 | 提示用户智能体不存在 |
| 500 | 服务器错误 | 提示用户稍后重试 |

### 错误处理示例

```typescript
async function safeGetAgentDetail(agentId: string) {
  try {
    const response = await fetch(
      `http://192.168.0.103:10089/v2/agent/${agentId}`
    );

    if (!response.ok) {
      if (response.status === 404) {
        throw new Error('智能体不存在');
      }
      throw new Error('服务器错误，请稍后重试');
    }

    const { success, data, message } = await response.json();

    if (!success) {
      throw new Error(message || '获取失败');
    }

    return data;
  } catch (error) {
    if (error instanceof Error) {
      console.error('获取智能体详情失败:', error.message);
      // 显示错误提示
      showErrorToast(error.message);
    }
    throw error;
  }
}
```

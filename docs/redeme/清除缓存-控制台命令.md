# 在浏览器控制台清除缓存

## 🚀 快速方法（推荐）

### 步骤1: 加载清除工具

在浏览器控制台（F12）中执行：

```javascript
// 加载清除工具脚本
const script = document.createElement('script');
script.src = '/clear-cache.js';
document.head.appendChild(script);
```

等待几秒后，会看到使用说明。

### 步骤2: 使用命令

```javascript
// 清除指定智能体缓存
await clearAgentCache('math_calculator')

// 或清除所有缓存
await clearAllAgentCache()

// 查看缓存列表
await listAgentCache()
```

### 步骤3: 刷新页面

按 F5 或执行：
```javascript
window.location.reload()
```

---

## 方法1: 清除指定智能体缓存（手动）

在浏览器控制台（F12）中执行：

```javascript
// 清除 math_calculator 的缓存
(async function() {
  const dbName = 'AgentUICache';
  const storeName = 'ui_bundles';
  const agentId = 'math_calculator';
  
  const request = indexedDB.open(dbName, 1);
  
  request.onsuccess = function(event) {
    const db = event.target.result;
    const transaction = db.transaction([storeName], 'readwrite');
    const objectStore = transaction.objectStore(storeName);
    const deleteRequest = objectStore.delete(agentId);
    
    deleteRequest.onsuccess = function() {
      console.log('✅ 缓存已清除:', agentId);
      alert('缓存已清除，请刷新页面');
    };
    
    deleteRequest.onerror = function() {
      console.error('❌ 清除失败');
    };
  };
})();
```

## 方法2: 清除所有缓存

```javascript
// 清除所有智能体的缓存
(async function() {
  const dbName = 'AgentUICache';
  const storeName = 'ui_bundles';
  
  const request = indexedDB.open(dbName, 1);
  
  request.onsuccess = function(event) {
    const db = event.target.result;
    const transaction = db.transaction([storeName], 'readwrite');
    const objectStore = transaction.objectStore(storeName);
    const clearRequest = objectStore.clear();
    
    clearRequest.onsuccess = function() {
      console.log('✅ 所有缓存已清除');
      alert('所有缓存已清除，请刷新页面');
    };
    
    clearRequest.onerror = function() {
      console.error('❌ 清除失败');
    };
  };
})();
```

## 方法3: 删除整个数据库

```javascript
// 删除整个 IndexedDB 数据库
(async function() {
  const dbName = 'AgentUICache';
  const deleteRequest = indexedDB.deleteDatabase(dbName);
  
  deleteRequest.onsuccess = function() {
    console.log('✅ 数据库已删除');
    alert('数据库已删除，请刷新页面');
  };
  
  deleteRequest.onerror = function() {
    console.error('❌ 删除失败');
  };
})();
```
